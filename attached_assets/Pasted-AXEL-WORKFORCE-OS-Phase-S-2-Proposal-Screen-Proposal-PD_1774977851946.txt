sql-- Proposal records — one per approved quote
CREATE TABLE proposals (
  id uuid PRIMARY KEY DEFAULT gen_random_uuid(),
  deal_id uuid REFERENCES deals(id) ON DELETE CASCADE,
  quote_id uuid REFERENCES quotes(id),
  created_by uuid REFERENCES auth.users(id),
  status text NOT NULL DEFAULT 'draft',
  -- status values: 'draft' | 'sent_to_client' | 'approved_proposal_requested' | 'underwriting_notified' | 'accepted' | 'declined'
  
  -- Pricing snapshot (pulled from quote/rating engine at time of proposal creation)
  wc_annual_premium numeric,
  wc_monthly_premium numeric,
  wfs_monthly_pepm numeric,
  wfs_annual_total numeric,
  total_monthly numeric,
  total_annual numeric,
  emod numeric,
  schedule_rating numeric,
  
  -- Rating breakdown (full JSON from R.2 engine)
  rating_breakdown jsonb,
  
  -- Proposal metadata
  effective_date date,
  expiration_date date,
  carrier_name text,
  program_name text,         -- e.g. 'Kind PEO Program'
  vertical_id text,
  
  -- PDF
  proposal_pdf_path text,    -- Supabase Storage path once generated
  
  -- Underwriting notification tracking
  uw_notified_at timestamptz,
  uw_notified_by uuid REFERENCES auth.users(id),
  uw_notification_trigger text, -- 'agent_client_request' | 'staff_manual'
  
  created_at timestamptz DEFAULT now(),
  updated_at timestamptz DEFAULT now()
);

-- Underwriting submission packages — bundles proposal + submission docs + loss history
CREATE TABLE underwriting_packages (
  id uuid PRIMARY KEY DEFAULT gen_random_uuid(),
  deal_id uuid REFERENCES deals(id) ON DELETE CASCADE,
  proposal_id uuid REFERENCES proposals(id),
  triggered_by uuid REFERENCES auth.users(id),
  trigger_type text NOT NULL,  -- 'agent_client_request' | 'staff_manual'
  status text NOT NULL DEFAULT 'pending',
  -- status: 'pending' | 'assembling' | 'sent' | 'failed'
  
  -- Assembled document paths
  documents jsonb,             -- Array of { type, label, storage_path }
  
  -- Email delivery
  email_sent_to text[],        -- Array of recipient emails
  email_sent_at timestamptz,
  email_message_id text,       -- For tracking/threading
  
  created_at timestamptz DEFAULT now(),
  updated_at timestamptz DEFAULT now()
);

-- Add proposal_id to bind_document_packages for cross-reference
ALTER TABLE bind_document_packages ADD COLUMN IF NOT EXISTS proposal_id uuid REFERENCES proposals(id);

-- Add proposal status to deals
ALTER TABLE deals ADD COLUMN IF NOT EXISTS proposal_status text DEFAULT 'none';
-- proposal_status: 'none' | 'draft' | 'sent_to_client' | 'approved_proposal_requested' | 'underwriting_notified'

-- Indexes
CREATE INDEX idx_proposals_deal ON proposals(deal_id);

javascriptconst { PDFDocument } = require('pdf-lib');
const { supabase } = require('../lib/supabase');
const fs = require('fs');
const path = require('path');

/**
 * Generates and stores the proposal PDF for a given proposal record.
 * Maps submission answers + proposal pricing fields into the AcroForm proposal template.
 * Template must be at: server/assets/pdf-templates/proposal.pdf
 */
async function generateProposalPdf(proposalId) {
  const { data: proposal, error: pError } = await supabase
    .from('proposals')
    .select('*')
    .eq('id', proposalId)
    .single();

  if (pError || !proposal) throw new Error('Proposal not found: ' + proposalId);

  // Load submission answers for this deal
  const { data: submission } = await supabase
    .from('submission_answers')
    .select('answers')
    .eq('deal_id', proposal.deal_id)
    .single();

  const answers = submission?.answers || {};

  // Build a unified field data map: submission answers + proposal pricing fields
  const fieldData = {
    ...answers,

    // Pricing fields — answer_keys used in form_field_mappings for document_type='proposal'
    proposal_wc_annual_premium: proposal.wc_annual_premium ? `$${Number(proposal.wc_annual_premium).toLocaleString('en-US', { minimumFractionDigits: 2 })}` : '',
    proposal_wc_monthly_premium: proposal.wc_monthly_premium ? `$${Number(proposal.wc_monthly_premium).toLocaleString('en-US', { minimumFractionDigits: 2 })}` : '',
    proposal_wfs_monthly_pepm: proposal.wfs_monthly_pepm ? `$${Number(proposal.wfs_monthly_pepm).toLocaleString('en-US', { minimumFractionDigits: 2 })}` : '',
    proposal_wfs_annual_total: proposal.wfs_annual_total ? `$${Number(proposal.wfs_annual_total).toLocaleString('en-US', { minimumFractionDigits: 2 })}` : '',
    proposal_total_monthly: proposal.total_monthly ? `$${Number(proposal.total_monthly).toLocaleString('en-US', { minimumFractionDigits: 2 })}` : '',
    proposal_total_annual: proposal.total_annual ? `$${Number(proposal.total_annual).toLocaleString('en-US', { minimumFractionDigits: 2 })}` : '',
    proposal_emod: proposal.emod ? String(proposal.emod) : '1.00',
    proposal_schedule_rating: proposal.schedule_rating ? String(proposal.schedule_rating) : '1.00',
    proposal_effective_date: proposal.effective_date ? formatDate(proposal.effective_date) : '',
    proposal_expiration_date: proposal.expiration_date ? formatDate(proposal.expiration_date) : '',
    proposal_carrier_name: proposal.carrier_name || '',
    proposal_program_name: proposal.program_name || '',
    proposal_date_prepared: formatDate(new Date().toISOString()),
  };

  // Load field mappings for proposal document type
  const { data: mappings, error: mError } = await supabase
    .from('form_field_mappings')
    .select('*')
    .eq('document_type', 'proposal');

  if (mError) throw new Error('Failed to load proposal field mappings: ' + mError.message);

  const templatePath = path.join(__dirname, '../assets/pdf-templates/proposal.pdf');
  if (!fs.existsSync(templatePath)) throw new Error('Proposal PDF template not found at server/assets/pdf-templates/proposal.pdf');

  const pdfBytes = fs.readFileSync(templatePath);
  const pdfDoc = await PDFDocument.load(pdfBytes);
  const form = pdfDoc.getForm();

  for (const mapping of mappings) {
    const rawValue = fieldData[mapping.answer_key];
    if (rawValue === undefined || rawValue === null) continue;

    let value = String(rawValue);
    if (mapping.transform === 'uppercase') value = value.toUpperCase();
    if (mapping.transform === 'boolean_yes_no') value = (rawValue === true || rawValue === 'true') ? 'Yes' : 'No';
    if (mapping.transform === 'date_mmddyyyy') value = formatDate(rawValue);

    try {
      form.getTextField(mapping.pdf_field_name).setText(value);
    } catch {
      console.warn(`Proposal PDF: field not found — ${mapping.pdf_field_name}`);
    }
  }

  form.flatten();
  const filledBytes = await pdfDoc.save();

  const storagePath = `proposals/${proposal.deal_id}/${proposalId}/proposal.pdf`;
  const { error: uploadError } = await supabase.storage
    .from('underwriting-documents')
    .upload(storagePath, filledBytes, { contentType: 'application/pdf', upsert: true });

  if (uploadError) throw new Error('Proposal PDF upload failed: ' + uploadError.message);

  // Update proposal record with storage path
  await supabase
    .from('proposals')
    .update({ proposal_pdf_path: storagePath, updated_at: new Date().toISOString() })
    .eq('id', proposalId);

  return storagePath;
}

function formatDate(dateStr) {
  try {
    const d = new Date(dateStr);
    return `${String(d.getMonth() + 1).padStart(2, '0')}/${String(d.getDate()).padStart(2, '0')}/${d.getFullYear()}`;
  } catch { return ''; }
}

 * Main entry point.
 * Assembles all submission docs, proposal PDF, and loss history into one package,
 * then sends an email to the UW alias with signed download links.
 */
async function sendUwSubmissionPackage(packageId) {
  try {
    // Load package record
    const { data: pkg, error: pkgError } = await supabase
      .from('underwriting_packages')
      .select('*, proposals(*)')
      .eq('id', packageId)
      .single();

    if (pkgError || !pkg) throw new Error('UW package not found: ' + packageId);

    const { deal_id, proposal_id } = pkg;
    const proposal = pkg.proposals;

    // Update status to assembling
    await supabase.from('underwriting_packages').update({ status: 'assembling' }).eq('id', packageId);

    // 1. Ensure proposal PDF exists — generate if not yet created
    let proposalPdfPath = proposal.proposal_pdf_path;
    if (!proposalPdfPath) {
      proposalPdfPath = await generateProposalPdf(proposal_id);
    }

    // 2. Load bind document package (submission PDFs)
    const { data: bindPkg } = await supabase
      .from('bind_document_packages')
      .select('documents')
      .eq('deal_id', deal_id)
      .order('created_at', { ascending: false })
      .limit(1)
      .single();

    // 3. Load loss history documents
    const { data: lossHistory } = await supabase
      .from('loss_history_documents')
      .select('file_name, storage_path, years_covered')
      .eq('deal_id', deal_id);

    // 4. Load deal info for email content
    const { data: deal } = await supabase
      .from('deals')
      .select('*, accounts(company_name)')
      .eq('id', deal_id)
      .single();

    // 5. Assemble document manifest
    const documents = [];

    // Proposal first
    documents.push({
      type: 'proposal',
      label: 'Axel Proposal',
      storage_path: proposalPdfPath
    });

    // Submission / bind docs
    if (bindPkg?.documents) {
      const submissionDocTypes = ['axel_supplemental', 'acord_130', 'carrier_supplemental', 'bind_order'];
      bindPkg.documents
        .filter(d => submissionDocTypes.includes(d.document_type))
        .forEach(d => documents.push(d));
    }

    // Loss history
    if (lossHistory && lossHistory.length > 0) {
      lossHistory.forEach(d => documents.push({
        type: 'loss_history',
        label: `Loss Run: ${d.years_covered || d.file_name}`,
        storage_path: d.storage_path
      }));
    }

    // 6. Generate signed URLs for each document (1-hour expiry)
    const docsWithUrls = await Promise.all(
      documents.map(async (doc) => {
        const { data: signedData, error } = await supabase.storage
          .from('underwriting-documents')
          .createSignedUrl(doc.storage_path, SIGNED_URL_EXPIRY);
        return {
          ...doc,
          signed_url: error ? null : signedData.signedUrl,
          url_error: error ? error.message : null
        };
      })
    );

    // 7. Build email HTML
    const clientName = deal?.accounts?.company_name || deal?.company_name || 'Client';
    const vertical = proposal.vertical_id ? capitalize(proposal.vertical_id) : 'WC';
    const dealUrl = `${APP_BASE_URL}/pipeline?deal=${deal_id}`;

    const docListHtml = docsWithUrls
      .map(d => d.signed_url
        ? `<tr>
            <td style="padding:8px 12px;border-bottom:1px solid #2a2a2a;color:#e0e0e0;font-size:14px;">${d.label}</td>
            <td style="padding:8px 12px;border-bottom:1px solid #2a2a2a;">
              <a href="${d.signed_url}" style="color:#E91E8C;text-decoration:none;font-size:13px;">Download PDF</a>
            </td>
          </tr>`
        : `<tr>
            <td style="padding:8px 12px;border-bottom:1px solid #2a2a2a;color:#888;font-size:14px;">${d.label} (unavailable)</td>
            <td></td>
          </tr>`
      ).join('');

    const emailHtml = `
<!DOCTYPE html>
<html>
<head><meta charset="utf-8"></head>
<body style="margin:0;padding:0;background:#0a0a0f;font-family:-apple-system,BlinkMacSystemFont,'Segoe UI',sans-serif;">

      <p style="color:#666;font-size:12px;margin:4px 0 0;">Underwriting Submission Package</p>
    </div>

    <div style="background:#111118;border:1px solid #1e1e2e;border-radius:12px;padding:28px;margin-bottom:24px;">
      <h2 style="color:#fff;font-size:20px;font-weight:600;margin:0 0 6px;">New Submission: ${clientName}</h2>

      <table style="width:100%;border-collapse:collapse;margin-bottom:20px;">
        <tr>
          <td style="color:#888;font-size:13px;padding:4px 0;width:160px;">Client</td>

function formatDateDisplay(dateStr) {
  try {
    return new Date(dateStr).toLocaleDateString('en-US', { month: 'short', day: 'numeric', year: 'numeric' });
  } catch { return ''; }
}

function capitalize(str) {
  return str ? str.charAt(0).toUpperCase() + str.slice(1) : '';
}

javascriptconst express = require('express');
const router = express.Router();
const { supabase } = require('../lib/supabase');
const { requireAuth, requireRole } = require('../middleware/auth');
const { generateProposalPdf } = require('../services/proposalPdfService');

// POST /api/proposals
// Create a proposal record from an approved quote — called by Axel staff
router.post('/', requireAuth, requireRole(['internal_sales', 'underwriter', 'admin']), async (req, res) => {
  const {
    deal_id, quote_id, wc_annual_premium, wc_monthly_premium,
    wfs_monthly_pepm, wfs_annual_total, total_monthly, total_annual,
    emod, schedule_rating, rating_breakdown, effective_date,
    expiration_date, carrier_name, program_name, vertical_id
  } = req.body;

  const { data: proposal, error } = await supabase
    .from('proposals')
    .insert({
      deal_id, quote_id, created_by: req.user.id,
      status: 'draft',
      wc_annual_premium, wc_monthly_premium,
      wfs_monthly_pepm, wfs_annual_total,
      total_monthly, total_annual,
      emod, schedule_rating, rating_breakdown,
      effective_date, expiration_date,
      carrier_name, program_name, vertical_id
    })
    .select()
    .single();

  if (error) return res.status(500).json({ error: error.message });

  await supabase.from('deals').update({ proposal_status: 'draft' }).eq('id', deal_id);
  await supabase.from('activity_log').insert({
    deal_id, actor_id: req.user.id,
    action: 'proposal_created',
    description: `Proposal created. WC Annual Premium: $${Number(wc_annual_premium).toLocaleString()}.`,
    metadata: { proposal_id: proposal.id }
  });

  // Generate proposal PDF asynchronously
  generateProposalPdf(proposal.id).catch(err => console.error('Proposal PDF gen failed:', err));

  res.json({ success: true, proposal });
});

// POST /api/proposals/:proposalId/request-approved-proposal
// Triggered by Agent, Client, Axel staff — sends UW submission package
router.post('/:proposalId/request-approved-proposal',
  requireAuth,
  requireRole(['agent', 'client', 'internal_sales', 'underwriter', 'admin']),
  async (req, res) => {
    const { proposalId } = req.params;

    const { data: proposal, error: pError } = await supabase
      .from('proposals')
      .select('*')
      .eq('id', proposalId)
      .single();

    if (pError || !proposal) return res.status(404).json({ error: 'Proposal not found.' });

    // Prevent duplicate sends
    if (proposal.status === 'underwriting_notified') {
      return res.status(400).json({ error: 'Underwriting has already been notified for this proposal.' });
    }

    // Determine trigger type
    const internalRoles = ['internal_sales', 'underwriter', 'admin'];
    const triggerType = internalRoles.includes(req.user.role) ? 'staff_manual' : 'agent_client_request';

    // Create UW package record
    const { data: uwPackage, error: uwError } = await supabase
      .from('underwriting_packages')
      .insert({
        deal_id: proposal.deal_id,
        proposal_id: proposalId,
        triggered_by: req.user.id,
        trigger_type: triggerType,
        status: 'pending'
      })
      .select()
      .single();

    if (uwError) return res.status(500).json({ error: uwError.message });

    // Update proposal status immediately
    await supabase.from('proposals')
      .update({ status: 'approved_proposal_requested', updated_at: new Date().toISOString() })
      .eq('id', proposalId);

    await supabase.from('deals')
      .update({ proposal_status: 'approved_proposal_requested' })
      .eq('id', proposal.deal_id);

    await supabase.from('activity_log').insert({
      deal_id: proposal.deal_id,
      actor_id: req.user.id,
      action: 'approved_proposal_requested',
      description: `Approved proposal requested by ${req.user.email}. Underwriting package assembly initiated.`,
      metadata: { proposal_id: proposalId, trigger_type: triggerType, uw_package_id: uwPackage.id }
    });

    // Fire async — return immediately
    sendUwSubmissionPackage(uwPackage.id)
      .catch(err => console.error('UW package send failed:', err));

    res.json({
      success: true,
      uwPackageId: uwPackage.id,
      message: 'Request received. Underwriting team will be notified shortly.'
    });
  }
);

// GET /api/proposals/:proposalId/uw-package-status
// Poll for UW package send status
router.get('/:proposalId/uw-package-status', requireAuth, async (req, res) => {
  const { data, error } = await supabase
    .from('underwriting_packages')
    .select('status, email_sent_at, email_sent_to, documents')
    .eq('proposal_id', req.params.proposalId)
    .order('created_at', { ascending: false })
    .limit(1)
    .single();

  if (error && error.code !== 'PGRST116') return res.status(500).json({ error: error.message });
  res.json({ package: data || null });
});
