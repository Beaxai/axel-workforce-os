AXEL WORKFORCE OS — Supplemental Phase S.4: Workforce Profile Widget Integration

CONTEXT
All prior phases (P1–P10, R.1–R.3, S.1–S.3) are complete. This phase integrates the Workforce Profile Widget into the existing quote flow. The widget replaces the current single-class-code input on the quote form with a multi-location, multi-class-code workforce profile screen that feeds structured payroll data into the rating engine.

OBJECTIVE
Insert the Workforce Profile Widget as a dedicated step in the quote flow between the business information screen and the premium calculation screen. The widget collects: locations, class codes per location, FT/PT employee counts, and annual payroll per code. On "Continue to rating" it passes a structured payload to the existing rating engine.

      id,
      street: prefill?.street || '',
      city: prefill?.city || '',
      state: prefill?.state || '',
      zip: prefill?.zip || '',
      confirmed: !!prefill,
      codes: {},
      search: '',
      vert: 'All',
      rows: [{ id: Date.now(), code: '', ft: '', pt: '', pay: '' }],
      pending: {},
    }]);
  }, [nextId]);

    setLocs(prev => prev.map(l => l.id === id ? { ...l, [field]: value } : l));
  };

  const addCode = (locId: number, code: string) => {
    setLocs(prev => prev.map(l => l.id === locId ? {
      ...l,
      codes: { ...l.codes, [code]: { ft: '', pt: '', pay: '' } },
      pending: { ...l.pending, [code]: true },
    } : l));
  };

    setLocs(prev => prev.map(l => {
      if (l.id !== locId) return l;
      const codes = { ...l.codes };
      const pending = { ...l.pending };
      delete codes[code];
      delete pending[code];
      return { ...l, codes, pending };

  async function handleAiAsk() {
    if (!aiInput.trim() || !modalCode || aiLoading) return;
    const q = aiInput.trim();
    setAiInput('');
    setAiLoading(true);
    const info = RICH[modalCode];
    const loc = locs.find(l => l.id === modalLocId);
    const prev = aiMessages[modalCode] || [];
    setAiMessages(m => ({ ...m, [modalCode!]: [...prev, { role: 'user', content: q }, { role: 'assistant', content: '...' }] }));
    try {
      const res = await fetch('/api/ai/classify', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({
          code: modalCode,
          name: info?.n,
          phraseology: info?.p,
          state: loc?.state,
          question: q,
          history: prev,
        }),
      });
      const data = await res.json();
      setAiMessages(m => ({
        ...m,
        [modalCode!]: [...(m[modalCode!] || []).filter(x => x.content !== '...'), { role: 'user', content: q }, { role: 'assistant', content: data.answer || 'Unable to answer.' }],
      }));
    } catch {
      setAiMessages(m => ({
        ...m,
        [modalCode!]: [...(m[modalCode!] || []).filter(x => x.content !== '...'), { role: 'user', content: q }, { role: 'assistant', content: 'Advisor unavailable.' }],
      }));
    }
    setAiLoading(false);
  }
};

tsimport Anthropic from '@anthropic-ai/sdk';

router.post('/api/ai/classify', async (req, res) => {
  const { code, name, phraseology, state, question, history } = req.body;
  
  const client = new Anthropic(); // uses ANTHROPIC_API_KEY env var
  
  const system = `You are a senior WC underwriting engineer and classification expert. 
Code: ${code} — ${name}. 
Phraseology: ${phraseology}. 
State: ${state || 'not specified'}. 
Answer in under 140 words. Plain text only.`;

  const messages = [
    ...(history || []).slice(-6).map((m: any) => ({ role: m.role, content: m.content })),
    { role: 'user' as const, content: question },
  ];

  const response = await client.messages.create({
    model: 'claude-sonnet-4-20250514',
    max_tokens: 1000,
    system,
    messages,
  });

  res.json({ answer: response.content[0].type === 'text' ? response.content[0].text : 'Unable to answer.' });
});
Confirm ANTHROPIC_API_KEY is already set in Railway environment variables — it should be from the existing AI features built in prior phases.
