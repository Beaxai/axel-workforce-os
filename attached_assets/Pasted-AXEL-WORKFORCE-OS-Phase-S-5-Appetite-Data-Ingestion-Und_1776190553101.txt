AXEL WORKFORCE OS — Phase S.5: Appetite Data Ingestion & Underwriting Appetite Engine

CONTEXT
All prior phases (P1–P10, R.1–R.3, S.1–S.4) are complete. This phase ingests Benchmark Insurance Company underwriting appetite data into Supabase and wires it into the platform so underwriters and internal staff can see write/refer/conditional/ineligible status for each class code at the point of quote and submission review.
Key facts about the data:

Source: Combined_Aligned_With_Benchmark_Appetite.xlsx — 25,196 rows, sheet Combined_All
Appetite is keyed on (state + class_code) — 75 codes have different determinations in different states, so state-level granularity is required
Four determination values: Acceptable, Referral, Conditional, Ineligible
24,323 rows have a determination; 873 are null (treat as Unknown)
UW_Considerations field contains binding conditions for Acceptable/Conditional codes and "Mandatory Referral - no field authority" for Referrals — max 318 chars, avg 84 chars
Existing rate_table already has rates from BIC.csv — this phase adds appetite as a separate but related table, joined on state + class_code


PART 1 — DATABASE SCHEMA
Run in Supabase SQL editor:
sql-- Appetite table: one row per state + class code
CREATE TABLE IF NOT EXISTS appetite (
  id uuid PRIMARY KEY DEFAULT gen_random_uuid(),
  state char(2) NOT NULL,
  class_code text NOT NULL,
  description text,
  base_rate numeric,
  uw_determination text NOT NULL DEFAULT 'Unknown',
  -- values: 'Acceptable' | 'Referral' | 'Conditional' | 'Ineligible' | 'Unknown'
  uw_considerations text,
  created_at timestamptz DEFAULT now(),
  updated_at timestamptz DEFAULT now(),
  UNIQUE(state, class_code)
);

-- Index for fast lookups by state + code (used at quote time)
CREATE INDEX IF NOT EXISTS idx_appetite_state_code ON appetite(state, class_code);

-- Index for filtering by determination (used in underwriting dashboards)
CREATE INDEX IF NOT EXISTS idx_appetite_determination ON appetite(uw_determination);

-- RLS: internal staff and underwriters can read; only service role can write
ALTER TABLE appetite ENABLE ROW LEVEL SECURITY;

CREATE POLICY "appetite_read" ON appetite
  FOR SELECT USING (true);

CREATE POLICY "appetite_write" ON appetite
  FOR ALL USING (auth.role() = 'service_role');

PART 2 — INGESTION SCRIPT
Create a new file at:
server/scripts/ingestAppetite.ts
typescriptimport * as XLSX from 'xlsx';
import { createClient } from '@supabase/supabase-js';
import path from 'path';

const supabase = createClient(
  process.env.SUPABASE_URL!,
  process.env.SUPABASE_SERVICE_KEY!
);

async function ingestAppetite() {
  const filePath = path.resolve(__dirname, '../../data/Combined_Aligned_With_Benchmark_Appetite.xlsx');
  console.log('Reading:', filePath);

  const workbook = XLSX.readFile(filePath);
  const sheet = workbook.Sheets['Combined_All'];
  const rows: any[] = XLSX.utils.sheet_to_json(sheet);
  console.log(`Rows read: ${rows.length}`);

  // Normalize determination value
  const normDet = (v: any): string => {
    if (!v) return 'Unknown';
    const s = String(v).trim();
    const map: Record<string, string> = {
      'acceptable': 'Acceptable',
      'referral': 'Referral',
      'conditional': 'Conditional',
      'ineligible': 'Ineligible',
    };
    return map[s.toLowerCase()] || 'Unknown';
  };

  const records = rows.map(r => ({
    state: String(r['State'] || '').trim().toUpperCase(),
    class_code: String(r['BaseCode'] || r['ClassCode'] || '').trim(),
    description: String(r['Description'] || '').trim() || null,
    base_rate: r['Base Rate'] != null ? parseFloat(r['Base Rate']) : null,
    uw_determination: normDet(r['Benchmark_UW_Determination']),
    uw_considerations: r['Benchmark_UW_Considerations']
      ? String(r['Benchmark_UW_Considerations']).trim()
      : null,
  })).filter(r => r.state && r.class_code);

  console.log(`Records to upsert: ${records.length}`);

  // Upsert in batches of 500
  const BATCH = 500;
  let inserted = 0;
  for (let i = 0; i < records.length; i += BATCH) {
    const batch = records.slice(i, i + BATCH);
    const { error } = await supabase
      .from('appetite')
      .upsert(batch, { onConflict: 'state,class_code' });
    if (error) {
      console.error(`Batch ${i}–${i + BATCH} error:`, error.message);
    } else {
      inserted += batch.length;
      console.log(`Upserted ${inserted}/${records.length}`);
    }
  }

  console.log('Done.');
}

ingestAppetite().catch(console.error);

PART 3 — BACKEND API ROUTE
Add the following to the Express backend (existing server/routes/ directory):
typescript// GET /api/appetite/:state/:classCode
// Returns appetite determination + considerations for a specific state/code
router.get('/api/appetite/:state/:classCode', async (req, res) => {
  const { state, classCode } = req.params;

  const { data, error } = await supabase
    .from('appetite')
    .select('uw_determination, uw_considerations, description, base_rate')
    .eq('state', state.toUpperCase())
    .eq('class_code', classCode)
    .single();

  if (error || !data) {
    return res.json({
      uw_determination: 'Unknown',
      uw_considerations: null,
      description: null,
      base_rate: null,
    });
  }

  res.json(data);
});

// POST /api/appetite/batch
// Returns appetite for multiple state/code pairs at once (used by WorkforceProfile)
router.post('/api/appetite/batch', async (req, res) => {
  const { lookups } = req.body as { lookups: Array<{ state: string; class_code: string }> };

  if (!lookups?.length) return res.json({ results: [] });

  // Fetch all at once using an OR filter
  const { data, error } = await supabase
    .from('appetite')
    .select('state, class_code, uw_determination, uw_considerations')
    .in('state', [...new Set(lookups.map(l => l.state.toUpperCase()))]);

  if (error) return res.status(500).json({ error: error.message });

  // Index by state+code for fast lookup
  const index: Record<string, any> = {};
  (data || []).forEach(r => { index[`${r.state}:${r.class_code}`] = r; });

  const results = lookups.map(l => ({
    state: l.state,
    class_code: l.class_code,
    ...(index[`${l.state.toUpperCase()}:${l.class_code}`] || { uw_determination: 'Unknown', uw_considerations: null }),
  }));

  res.json({ results });
});

API route for this page:
typescript// GET /api/appetite?state=CA&determination=Referral&search=roofing&page=0&limit=50
router.get('/api/appetite', async (req, res) => {
  const { state, determination, search, page = '0', limit = '50' } = req.query as Record<string, string>;
  const from = parseInt(page) * parseInt(limit);
  const to = from + parseInt(limit) - 1;

  let query = supabase
    .from('appetite')
    .select('*', { count: 'exact' })
    .order('state')
    .order('class_code')
    .range(from, to);

  if (state) query = query.eq('state', state.toUpperCase());
  if (determination) query = query.eq('uw_determination', determination);
  if (search) query = query.ilike('description', `%${search}%`);

  const { data, count, error } = await query;
  if (error) return res.status(500).json({ error: error.message });
  res.json({ data, total: count });
});
