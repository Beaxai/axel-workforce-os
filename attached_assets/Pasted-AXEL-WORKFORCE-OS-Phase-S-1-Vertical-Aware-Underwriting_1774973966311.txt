AXEL WORKFORCE OS — Phase S.1: Vertical-Aware Underwriting Submission Engine

CONTEXT
You are building on the Axel Workforce OS platform. The confirmed tech stack is React (frontend), Node.js (backend), Supabase (database), and Railway (hosting). The UI design system is dark glassmorphism: base background #060608, sole accent color #E91E8C (solid, no gradients), frosted glass panels using rgba(255,255,255,0.05) background with backdrop-filter: blur(12px) and 1px solid rgba(255,255,255,0.08) borders. All icons use Lucide React exclusively. No other icon libraries. No external UI component libraries.
This phase builds the complete vertical-aware underwriting submission flow, loss history document upload, and bind document package generation for the WC-only product path.

PART 1 — DATABASE SCHEMA (Supabase)
Run these migrations in Supabase SQL editor.
sql-- Vertical question sets registry
CREATE TABLE submission_question_sets (
  id uuid PRIMARY KEY DEFAULT gen_random_uuid(),
  vertical_id text NOT NULL,           -- e.g. 'cannabis', 'construction', 'hospitality'
  vertical_label text NOT NULL,        -- Display name
  version integer NOT NULL DEFAULT 1,
  is_active boolean NOT NULL DEFAULT true,
  created_at timestamptz DEFAULT now()
);

-- Individual questions belonging to a question set
CREATE TABLE submission_questions (
  id uuid PRIMARY KEY DEFAULT gen_random_uuid(),
  question_set_id uuid REFERENCES submission_question_sets(id) ON DELETE CASCADE,
  section text NOT NULL,               -- e.g. 'Business Information', 'Operations', 'Loss History'
  display_order integer NOT NULL,
  answer_key text NOT NULL,            -- Snake_case unique key, used for PDF field mapping
  question_text text NOT NULL,
  help_text text,
  input_type text NOT NULL,            -- 'text' | 'number' | 'select' | 'multiselect' | 'boolean' | 'date' | 'textarea'
  options jsonb,                       -- For select/multiselect: ["Option A","Option B"]
  is_required boolean DEFAULT true,
  conditional_on_key text,             -- answer_key of another question
  conditional_on_value text,           -- Show this question only when that key equals this value
  created_at timestamptz DEFAULT now()
);

-- Stored submission answers per deal
CREATE TABLE submission_answers (
  id uuid PRIMARY KEY DEFAULT gen_random_uuid(),
  deal_id uuid REFERENCES deals(id) ON DELETE CASCADE,
  quote_id uuid REFERENCES quotes(id),
  question_set_id uuid REFERENCES submission_question_sets(id),
  answers jsonb NOT NULL DEFAULT '{}', -- { answer_key: value } flat map
  submitted_by uuid REFERENCES auth.users(id),
  submitted_at timestamptz,
  status text NOT NULL DEFAULT 'draft', -- 'draft' | 'submitted' | 'bind_requested'
  created_at timestamptz DEFAULT now(),
  updated_at timestamptz DEFAULT now()
);

-- PDF field mapping: answer_key → field name in each of the 4 bind documents
CREATE TABLE form_field_mappings (
  id uuid PRIMARY KEY DEFAULT gen_random_uuid(),
  answer_key text NOT NULL,
  document_type text NOT NULL,         -- 'axel_supplemental' | 'acord_130' | 'carrier_supplemental' | 'bind_order'
  pdf_field_name text NOT NULL,        -- Exact AcroForm field name in that PDF
  transform text,                      -- Optional: 'uppercase' | 'date_mmddyyyy' | 'boolean_yes_no'
  created_at timestamptz DEFAULT now()
);

-- Loss history documents uploaded per deal
CREATE TABLE loss_history_documents (
  id uuid PRIMARY KEY DEFAULT gen_random_uuid(),
  deal_id uuid REFERENCES deals(id) ON DELETE CASCADE,
  quote_id uuid REFERENCES quotes(id),
  file_name text NOT NULL,
  storage_path text NOT NULL,          -- Supabase Storage path
  file_size_bytes integer,
  uploaded_by uuid REFERENCES auth.users(id),
  uploaded_at timestamptz DEFAULT now(),
  years_covered text,                  -- e.g. '2022-2024'
  notes text
);

-- Bind document package — one record per bind request
CREATE TABLE bind_document_packages (
  id uuid PRIMARY KEY DEFAULT gen_random_uuid(),
  deal_id uuid REFERENCES deals(id) ON DELETE CASCADE,
  quote_id uuid REFERENCES quotes(id),
  requested_by uuid REFERENCES auth.users(id),
  requested_at timestamptz DEFAULT now(),
  status text NOT NULL DEFAULT 'generating', -- 'generating' | 'pending_signature' | 'signed' | 'failed'
  hellosign_signature_request_id text,
  documents jsonb,                     -- Array of { document_type, storage_path, hellosign_file_id }
  loss_history_included boolean DEFAULT false,
  activity_log_entry_id uuid,
  created_at timestamptz DEFAULT now(),
  updated_at timestamptz DEFAULT now()
);

-- Indexes
CREATE INDEX idx_submission_answers_deal ON submission_answers(deal_id);
CREATE INDEX idx_submission_questions_set ON submission_questions(question_set_id, display_order);
CREATE INDEX idx_form_field_mappings_key ON form_field_mappings(answer_key, document_type);
CREATE INDEX idx_loss_history_deal ON loss_history_documents(deal_id);
CREATE INDEX idx_bind_packages_deal ON bind_document_packages(deal_id);

javascriptconst express = require('express');
const router = express.Router();
const { supabase } = require('../lib/supabase');
const { requireAuth, requireRole } = require('../middleware/auth');
const { generateBindPackage } = require('../services/bindDocumentService');

// GET /api/submission/question-set/:verticalId
// Returns the active question set + all questions for a vertical
router.get('/question-set/:verticalId', requireAuth, async (req, res) => {
  const { verticalId } = req.params;

  const { data: questionSet, error: qsError } = await supabase
    .from('submission_question_sets')
    .select('*')
    .eq('vertical_id', verticalId)
    .eq('is_active', true)
    .single();

  if (qsError || !questionSet) {
    return res.status(404).json({ error: 'Question set not found for vertical: ' + verticalId });
  }

  const { data: questions, error: qError } = await supabase
    .from('submission_questions')
    .select('*')
    .eq('question_set_id', questionSet.id)
    .order('display_order');

  if (qError) return res.status(500).json({ error: qError.message });

  res.json({ questionSet, questions });
});

// GET /api/submission/answers/:dealId
// Returns saved draft answers for a deal
router.get('/answers/:dealId', requireAuth, async (req, res) => {
  const { dealId } = req.params;

  const { data, error } = await supabase
    .from('submission_answers')
    .select('*')
    .eq('deal_id', dealId)
    .order('created_at', { ascending: false })
    .limit(1)
    .single();

  if (error && error.code !== 'PGRST116') return res.status(500).json({ error: error.message });

  res.json({ answers: data || null });
});

// POST /api/submission/answers/:dealId
// Save or update draft answers (upsert)
router.post('/answers/:dealId', requireAuth, async (req, res) => {
  const { dealId } = req.params;
  const { answers, questionSetId, quoteId, status } = req.body;

  // Check if record exists
  const { data: existing } = await supabase
    .from('submission_answers')
    .select('id')
    .eq('deal_id', dealId)
    .single();

  let result;
  if (existing) {
    result = await supabase
      .from('submission_answers')
      .update({ answers, status: status || 'draft', updated_at: new Date().toISOString() })
      .eq('id', existing.id)
      .select()
      .single();
  } else {
    result = await supabase
      .from('submission_answers')
      .insert({
        deal_id: dealId,
        quote_id: quoteId,
        question_set_id: questionSetId,
        answers,
        submitted_by: req.user.id,
        status: status || 'draft'
      })
      .select()
      .single();
  }

  if (result.error) return res.status(500).json({ error: result.error.message });

  // Update deal submission_status
  await supabase
    .from('deals')
    .update({ submission_status: status || 'in_progress' })
    .eq('id', dealId);

  // Auto-log to activity feed
  await supabase.from('activity_log').insert({
    deal_id: dealId,
    actor_id: req.user.id,
    action: status === 'submitted' ? 'submission_completed' : 'submission_updated',
    description: status === 'submitted' ? 'Submission application completed and submitted.' : 'Submission application saved as draft.',
    metadata: { question_set_id: questionSetId }
  });

  res.json({ success: true, record: result.data });
});

// POST /api/submission/request-bind/:dealId
// Triggers bind document package generation
// Allowed roles: agent, client, internal_sales, underwriter, admin
router.post('/request-bind/:dealId', requireAuth, requireRole(['agent','client','internal_sales','underwriter','admin']), async (req, res) => {
  const { dealId } = req.params;
  const { quoteId } = req.body;

  // Verify submission is in 'submitted' status
  const { data: submission, error: subError } = await supabase
    .from('submission_answers')
    .select('*')
    .eq('deal_id', dealId)
    .single();

  if (subError || !submission) {
    return res.status(400).json({ error: 'No completed submission found for this deal.' });
  }

  if (submission.status !== 'submitted') {
    return res.status(400).json({ error: 'Submission must be completed before requesting bind.' });
  }

  // Check for loss history docs
  const { data: lossHistory } = await supabase
    .from('loss_history_documents')
    .select('id')
    .eq('deal_id', dealId);

  const hasLossHistory = lossHistory && lossHistory.length > 0;

  // Create bind package record
  const { data: bindPackage, error: bpError } = await supabase
    .from('bind_document_packages')
    .insert({
      deal_id: dealId,
      quote_id: quoteId,
      requested_by: req.user.id,
      status: 'generating',
      loss_history_included: hasLossHistory
    })
    .select()
    .single();

  if (bpError) return res.status(500).json({ error: bpError.message });

  // Update deal and submission status
  await supabase.from('deals').update({ submission_status: 'bind_requested' }).eq('id', dealId);
  await supabase.from('submission_answers').update({ status: 'bind_requested' }).eq('id', submission.id);

  // Auto-log activity
  await supabase.from('activity_log').insert({
    deal_id: dealId,
    actor_id: req.user.id,
    action: 'bind_requested',
    description: `Bind requested by ${req.user.email}. Document package generation initiated.`,
    metadata: { bind_package_id: bindPackage.id, loss_history_included: hasLossHistory }
  });

  // Trigger async document generation (do not await — return immediately)
  generateBindPackage(bindPackage.id, dealId, quoteId, submission.answers)
    .catch(err => console.error('Bind package generation failed:', err));

  res.json({ success: true, bindPackageId: bindPackage.id, message: 'Bind request received. Document package is being generated.' });
});

module.exports = router;

router.post('/:dealId/upload', requireAuth, upload.single('file'), async (req, res) => {
  const { dealId } = req.params;
  const { yearsCovered, notes } = req.body;

  if (!req.file) return res.status(400).json({ error: 'No file provided.' });
  if (req.file.mimetype !== 'application/pdf') return res.status(400).json({ error: 'Only PDF files are accepted for loss history.' });

  const storagePath = `loss-history/${dealId}/${Date.now()}_${req.file.originalname}`;

  const { error: uploadError } = await supabase.storage
    .from('underwriting-documents')
    .upload(storagePath, req.file.buffer, { contentType: 'application/pdf', upsert: false });

  if (uploadError) return res.status(500).json({ error: uploadError.message });

  const { data, error: dbError } = await supabase
    .from('loss_history_documents')
    .insert({
      deal_id: dealId,
      file_name: req.file.originalname,
      storage_path: storagePath,
      file_size_bytes: req.file.size,
      uploaded_by: req.user.id,
      years_covered: yearsCovered || null,
      notes: notes || null
    })
    .select()
    .single();

  if (dbError) return res.status(500).json({ error: dbError.message });

  await supabase.from('activity_log').insert({
    deal_id: dealId,
    actor_id: req.user.id,
    action: 'loss_history_uploaded',
    description: `Loss run document uploaded: ${req.file.originalname}`,
    metadata: { storage_path: storagePath, years_covered: yearsCovered }
  });

  res.json({ success: true, document: data });
});

// DELETE /api/loss-history/:dealId/:docId
router.delete('/:dealId/:docId', requireAuth, async (req, res) => {

javascriptconst { PDFDocument } = require('pdf-lib');
const { supabase } = require('../lib/supabase');
const fs = require('fs');
const path = require('path');

// Document type keys — must match filenames in server/assets/pdf-templates/
const DOCUMENT_TYPES = [
  { key: 'axel_supplemental', label: 'Axel Supplemental Application', filename: 'axel_supplemental.pdf' },
  { key: 'acord_130', label: 'ACORD 130 Workers Compensation Application', filename: 'acord_130.pdf' },
  { key: 'carrier_supplemental', label: 'Carrier Supplemental Application', filename: 'carrier_supplemental.pdf' },
  { key: 'bind_order', label: 'Bind Order Form', filename: 'bind_order.pdf' }
];

async function generateBindPackage(bindPackageId, dealId, quoteId, answers) {
  try {
    // Load field mappings for all document types
    const { data: mappings, error: mappingError } = await supabase
      .from('form_field_mappings')
      .select('*');

    if (mappingError) throw new Error('Failed to load field mappings: ' + mappingError.message);

    const generatedDocs = [];

    for (const docType of DOCUMENT_TYPES) {
      const templatePath = path.join(__dirname, '../assets/pdf-templates', docType.filename);

      if (!fs.existsSync(templatePath)) {
        console.warn(`PDF template not found: ${templatePath} — skipping.`);
        continue;
      }

      const pdfBytes = fs.readFileSync(templatePath);
      const pdfDoc = await PDFDocument.load(pdfBytes);
      const form = pdfDoc.getForm();

      // Get mappings for this document type
      const docMappings = mappings.filter(m => m.document_type === docType.key);

      for (const mapping of docMappings) {
        const rawValue = answers[mapping.answer_key];
        if (rawValue === undefined || rawValue === null) continue;

        let value = String(rawValue);

        // Apply transforms
        if (mapping.transform === 'uppercase') value = value.toUpperCase();
        if (mapping.transform === 'boolean_yes_no') value = rawValue === true || rawValue === 'true' ? 'Yes' : 'No';
        if (mapping.transform === 'date_mmddyyyy') {
          try {
            const d = new Date(rawValue);
            value = `${String(d.getMonth()+1).padStart(2,'0')}/${String(d.getDate()).padStart(2,'0')}/${d.getFullYear()}`;
          } catch { /* leave as-is */ }
        }

        try {
          const field = form.getTextField(mapping.pdf_field_name);
          field.setText(value);
        } catch {
          // Field name mismatch — log but do not fail the entire package
          console.warn(`Field not found in ${docType.key}: ${mapping.pdf_field_name}`);
        }
      }

      // Flatten form (makes fields non-editable — appropriate for legal submission copy)
      form.flatten();

      const filledPdfBytes = await pdfDoc.save();

      // Upload to Supabase Storage
      const storagePath = `bind-packages/${dealId}/${bindPackageId}/${docType.filename}`;
      const { error: uploadError } = await supabase.storage
        .from('underwriting-documents')
        .upload(storagePath, filledPdfBytes, { contentType: 'application/pdf', upsert: true });

      if (uploadError) throw new Error('Upload failed for ' + docType.key + ': ' + uploadError.message);

      generatedDocs.push({ document_type: docType.key, label: docType.label, storage_path: storagePath });
    }

    // Attach loss history docs to package record
    const { data: lossHistory } = await supabase
      .from('loss_history_documents')
      .select('file_name, storage_path')
      .eq('deal_id', dealId);

    if (lossHistory && lossHistory.length > 0) {
      lossHistory.forEach(doc => {
        generatedDocs.push({ document_type: 'loss_history', label: doc.file_name, storage_path: doc.storage_path });
      });
    }

    // Update bind package record to pending_signature
    await supabase
      .from('bind_document_packages')
      .update({
        status: 'pending_signature',
        documents: generatedDocs,
        updated_at: new Date().toISOString()
      })
      .eq('id', bindPackageId);

    // Auto-log completion
    await supabase.from('activity_log').insert({
      deal_id: dealId,
      action: 'bind_package_ready',
      description: `Bind document package generated. ${generatedDocs.length} documents ready for signature.`,
      metadata: { bind_package_id: bindPackageId, document_count: generatedDocs.length }
    });

  } catch (err) {
    console.error('generateBindPackage error:', err);
    await supabase
      .from('bind_document_packages')
      .update({ status: 'failed', updated_at: new Date().toISOString() })
      .eq('id', bindPackageId);

    await supabase.from('activity_log').insert({
      deal_id: dealId,
      action: 'bind_package_failed',
      description: `Bind document package generation failed: ${err.message}`,
      metadata: { bind_package_id: bindPackageId }
    });
  }
}

  const [result, setResult] = useState(null); // 'success' | 'error'
  const [message, setMessage] = useState('');

  const canBind = BIND_ALLOWED_ROLES.includes(userRole);
  const isReady = submissionStatus === 'submitted';
  const alreadyRequested = submissionStatus === 'bind_requested';

  async function handleRequestBind() {
    if (!canBind || !isReady) return;
    setLoading(true);
    setResult(null);

    const res = await fetch(`/api/submission/request-bind/${dealId}`, {
      method: 'POST',
      headers: { 'Content-Type': 'application/json', Authorization: `Bearer ${localStorage.getItem('token')}` },
      body: JSON.stringify({ quoteId })
    });

    const data = await res.json();
    setLoading(false);

    if (data.success) {
      setResult('success');
      setMessage('Bind request submitted. Document package is being prepared.');
      if (onSuccess) onSuccess(data.bindPackageId);
    } else {
      setResult('error');
      setMessage(data.error || 'Request failed. Please try again.');
    }
  }

  if (alreadyRequested) {
    return (
      <div style={{ display: 'flex', alignItems: 'center', gap: '8px', color: '#4caf50', fontSize: '14px' }}>
        <CheckCircle size={16} /> Bind Requested — Document package in preparation
      </div>
    );
  }

  if (!canBind) return null;

  return (
    <div>
      <button
        onClick={handleRequestBind}
        disabled={!isReady || loading}
        style={{
          display: 'flex', alignItems: 'center', gap: '8px',
          padding: '12px 28px', borderRadius: '8px',
          border: 'none',
          background: isReady ? accent : 'rgba(255,255,255,0.08)',
          color: isReady ? '#fff' : 'rgba(255,255,255,0.3)',
          cursor: isReady && !loading ? 'pointer' : 'not-allowed',
          fontSize: '15px', fontWeight: '600'
        }}>
        {loading ? <Loader size={16} style={{ animation: 'spin 1s linear infinite' }} /> : <FileSignature size={16} />}
        {loading ? 'Submitting...' : 'Request to Bind Coverage'}
      </button>
      {!isReady && !alreadyRequested && (
        <p style={{ color: 'rgba(255,255,255,0.35)', fontSize: '12px', marginTop: '6px' }}>
          Complete and submit the application before requesting bind.
        </p>
      )}
      {result && (
        <p style={{ color: result === 'success' ? '#4caf50' : '#ff4d4f', fontSize: '13px', marginTop: '8px', display: 'flex', alignItems: 'center', gap: '6px' }}>
          {result === 'success' ? <CheckCircle size={13} /> : <AlertCircle size={13} />} {message}
        </p>
      )}
    </div>
  );
}
